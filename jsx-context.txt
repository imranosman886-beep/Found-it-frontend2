import ProtectedRoute from "./components/ProtectedRoute";
import Navbar from "./components/Navbar";
import Footer from "./components/Footer";

import Home from "./pages/Home";
import Login from "./pages/Login";
import Signup from "./pages/Signup";
import ReportLost from "./pages/ReportLost";
import ReportFound from "./pages/ReportFound";
import ItemDetails from "./pages/ItemDetails";
import BrowseItems from "./pages/BrowseItems";
import MyReports from "./pages/MyReports";
import MyClaims from "./pages/MyClaims";
import Notifications from "./pages/Notifications";
import AdminDashboard from "./pages/AdminDashboard";
import Profile from "./pages/Profile";

function Page({ title }) {
  return (
    <main className="page-container">
      <h1>{title}</h1>
      <p>Welcome to FoundIt. This page is being built.</p>
    </main>
  );
}

function App() {
  return (
    <AuthProvider>
      <BrowserRouter>
        <Navbar />

        <Routes>
          <Route path="/" element={<Home />} />
          <Route path="/login" element={<Login />} />

          <Route path="/items/:id" element={<ItemDetails />} />

          <Route path="*" element={<Page title="404 - Page Not Found" />} />

export default App;

import { Link, NavLink, useNavigate } from "react-router-dom";
import { Bell, LogOut, Menu, X } from "lucide-react";
import { useState } from "react";
import { useAuth } from "../context/AuthContext";

function Navbar() {
  const { user, logoutUser } = useAuth();
  const [menuOpen, setMenuOpen] = useState(false);
  const navigate = useNavigate();

  async function handleLogout() {
    try {
      await logoutUser();
      setMenuOpen(false);
      navigate("/");
    } catch (error) {
      alert(error.message);
    }
  }

  const linkClass = ({ isActive }) =>
    `nav-link ${isActive ? "active" : ""}`;

  function closeMenu() {
    setMenuOpen(false);
  }

  return (
    <header className="navbar">
      <div className="navbar-container">
        <Link to="/" className="logo" onClick={closeMenu}>
          Found<span>It</span>
        </Link>

        <button
          className="menu-toggle"
          onClick={() => setMenuOpen(!menuOpen)}
          aria-label="Toggle navigation"
        >
          {menuOpen ? <X /> : <Menu />}
        </button>

        <nav className={`nav-links ${menuOpen ? "open" : ""}`}>
          <NavLink to="/" className={linkClass} onClick={closeMenu}>
            Home
          </NavLink>

          <NavLink to="/items" className={linkClass} onClick={closeMenu}>
            Browse Items
          </NavLink>

          {user && (
            <>
              <NavLink to="/report-lost" className={linkClass} onClick={closeMenu}>
                Report Lost
              </NavLink>

              <NavLink to="/report-found" className={linkClass} onClick={closeMenu}>
                Report Found
              </NavLink>

              <NavLink to="/my-reports" className={linkClass} onClick={closeMenu}>
                My Reports
              </NavLink>

              <NavLink to="/my-claims" className={linkClass} onClick={closeMenu}>
                My Claims
              </NavLink>

              <NavLink to="/profile" className={linkClass} onClick={closeMenu}>
                My Profile
              </NavLink>

              <NavLink to="/notifications" className={linkClass} onClick={closeMenu}>
                <Bell size={17} /> Notifications
              </NavLink>
            </>
          )}

          {user?.role === "admin" && (
            <NavLink to="/admin" className={linkClass} onClick={closeMenu}>
              Admin
            </NavLink>

          {!user ? (
            <div className="nav-actions">
              <Link to="/login" className="login-link" onClick={closeMenu}>
                Login
              </Link>

              <Link to="/signup" className="signup-button" onClick={closeMenu}>

          ) : (
            <div className="nav-actions">
              <button className="logout-button" onClick={handleLogout}>
                <LogOut size={16} /> Logout
              </button>
            </div>
          )}

      </div>
    </header>
  );
}

import { Navigate, Outlet, useLocation } from "react-router-dom";
import { useAuth } from "../context/AuthContext";

function ProtectedRoute({ adminOnly = false }) {
  const { user, loading, isAdmin } = useAuth();
  const location = useLocation();

  if (loading) {
    return <p className="loading-message">Loading FoundIt...</p>;
  }

  if (!user) {
    return (
      <Navigate
        to="/login"
        state={{ from: location }}
        replace
      />
    );
  }

  if (adminOnly && !isAdmin) {
    return <Navigate to="/" replace />;
  }

  return <Outlet />;
}

export default ProtectedRoute;

import { createContext, useContext, useEffect, useState } from "react";
import * as authService from "../services/authService";

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function loadUser() {
      try {
        const data = await authService.checkSession();

        if (data.logged_in && data.user) {
          setUser(data.user);
        } else {
          setUser(null);
        }
      } catch (error) {
        setUser(null);
      } finally {
        setLoading(false);
      }
    }

    loadUser();
  }, []);

  async function loginUser(identifier, password, isAdmin = false) {
    const data = await authService.login(identifier, password, isAdmin);
    setUser(data.user);
    return data;
  }

  async function signupUser(userDetails) {
    const data = await authService.signup(userDetails);

    // Save the newly registered user as authenticated.
    setUser(data.user);

    return data;
  }

  async function logoutUser() {
    await authService.logout();
    setUser(null);
  }

  const value = {
    user,
    loading,
    isAuthenticated: Boolean(user),
    isAdmin: user?.role === "admin",
    loginUser,
    signupUser,
    logoutUser,
  };

  return (
    <AuthContext.Provider value={value}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  return useContext(AuthContext);
}

import { useEffect, useState } from "react";
import {
  Users,
  Package,
  Search,
  LayoutDashboard,
  RefreshCw,
  ClipboardList,
  Check,
  X,
  Handshake,
  Trash2,
} from "lucide-react";

import { useAuth } from "../context/AuthContext";

import {
  getAdminUsers,
  getAdminLostItems,
  getAdminFoundItems,
  updateUserRole,
  updateLostItemStatus,
  updateFoundItemStatus,
  deleteLostItem,
  deleteFoundItem,
  getAdminClaims,
  updateClaimStatus,
  completeClaimHandover,
} from "../services/adminService";

function AdminDashboard() {
  const { user: currentUser } = useAuth();

  const [activeTab, setActiveTab] = useState("overview");
  const [users, setUsers] = useState([]);
  const [lostItems, setLostItems] = useState([]);
  const [foundItems, setFoundItems] = useState([]);
  const [claims, setClaims] = useState([]);

  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");
  const [actionLoading, setActionLoading] = useState(null);

  async function loadDashboard() {
    setLoading(true);
    setError("");

    try {
      const [usersData, lostData, foundData, claimsData] =
        await Promise.all([
          getAdminUsers(),
          getAdminLostItems(),
          getAdminFoundItems(),
          getAdminClaims(),
        ]);

      setUsers(usersData);
      setLostItems(lostData);
      setFoundItems(foundData);
      setClaims(claimsData);
    } catch (err) {
      setError(err.message || "Could not load admin dashboard.");
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => {
    loadDashboard();
  }, []);

  async function handleRoleChange(userId, newRole) {
    setError("");
    setSuccess("");
    setActionLoading(`user-${userId}`);

    try {
      await updateUserRole(userId, newRole);

      setUsers((currentUsers) =>
        currentUsers.map((user) =>
          user.id === userId ? { ...user, role: newRole } : user
        )
      );

      setSuccess("User role updated successfully.");
    } catch (err) {
      setError(err.message || "Could not update user role.");
    } finally {
      setActionLoading(null);
    }
  }

  async function handleLostStatusChange(itemId, status) {
    setError("");
    setSuccess("");
    setActionLoading(`lost-${itemId}`);

    try {
      await updateLostItemStatus(itemId, status);

      setLostItems((currentItems) =>
        currentItems.map((item) =>
          item.id === itemId ? { ...item, status } : item
        )
      );

      setSuccess("Lost report status updated successfully.");
    } catch (err) {
      setError(err.message || "Could not update lost report.");
    } finally {
      setActionLoading(null);
    }
  }

  async function handleFoundStatusChange(itemId, status) {
    setError("");
    setSuccess("");
    setActionLoading(`found-${itemId}`);

    try {
      await updateFoundItemStatus(itemId, status);

      setFoundItems((currentItems) =>
        currentItems.map((item) =>
          item.id === itemId ? { ...item, status } : item
        )
      );

      setSuccess("Found report status updated successfully.");
    } catch (err) {
      setError(err.message || "Could not update found report.");
    } finally {
      setActionLoading(null);
    }
  }

  async function handleDeleteLostItem(itemId) {
    const confirmed = window.confirm(
      "Are you sure you want to delete this lost report? This cannot be undone."
    );

    if (!confirmed) return;

    setError("");
    setSuccess("");
    setActionLoading(`delete-lost-${itemId}`);

    try {
      await deleteLostItem(itemId);

      setLostItems((currentItems) =>
        currentItems.filter((item) => item.id !== itemId)
      );

      setSuccess("Lost report deleted successfully.");
    } catch (err) {
      setError(err.message || "Could not delete lost report.");
    } finally {
      setActionLoading(null);
    }
  }

  async function handleDeleteFoundItem(itemId) {
    const confirmed = window.confirm(
      "Are you sure you want to delete this found report? This cannot be undone."
    );

    if (!confirmed) return;

    setError("");
    setSuccess("");
    setActionLoading(`delete-found-${itemId}`);

    try {
      await deleteFoundItem(itemId);

      setFoundItems((currentItems) =>
        currentItems.filter((item) => item.id !== itemId)
      );

      setSuccess("Found report deleted successfully.");
    } catch (err) {
      setError(err.message || "Could not delete found report.");
    } finally {
      setActionLoading(null);
    }
  }

  async function handleClaimStatus(claimId, status) {
    const action = status === "approved" ? "approve" : "reject";

    if (!window.confirm(`Are you sure you want to ${action} this claim?`)) {
      return;
    }

    setError("");
    setSuccess("");
    setActionLoading(`claim-${claimId}`);

    try {
      await updateClaimStatus(claimId, status);
      await loadDashboard();
      setSuccess(`Claim ${action}d successfully.`);
    } catch (err) {
      setError(err.message || `Could not ${action} claim.`);
    } finally {
      setActionLoading(null);
    }
  }

  async function handleHandover(claimId) {
    if (!window.confirm("Confirm that this item has been handed over?")) {
      return;
    }

    setError("");
    setSuccess("");
    setActionLoading(`handover-${claimId}`);

    try {
      await completeClaimHandover(claimId);
      await loadDashboard();
      setSuccess("Handover recorded successfully.");
    } catch (err) {
      setError(err.message || "Could not record handover.");
    } finally {
      setActionLoading(null);
    }
  }

  const tabs = [
    { id: "overview", label: "Overview", icon: LayoutDashboard },
    { id: "users", label: "Users", icon: Users },
    { id: "lost", label: "Lost Reports", icon: Search },
    { id: "found", label: "Found Reports", icon: Package },
    { id: "claims", label: "Claims", icon: ClipboardList },
  ];

  if (loading) {
    return <p className="loading-message">Loading admin dashboard...</p>;
  }

  return (
    <main className="page-container admin-page">
      <div className="admin-heading">
        <div>
          <h1>Admin Dashboard</h1>
          <p>Manage users, reports and claims.</p>
        </div>

        <button
          className="admin-refresh-button"
          onClick={loadDashboard}
          type="button"
        >
          <RefreshCw size={17} />
          Refresh
        </button>
      </div>

      {error && <p className="form-error">{error}</p>}
      {success && <p className="form-success">{success}</p>}

      <div className="admin-tabs">
        {tabs.map((tab) => {
          const Icon = tab.icon;

          return (
            <button
              key={tab.id}
              type="button"
              className={activeTab === tab.id ? "active" : ""}
              onClick={() => setActiveTab(tab.id)}
            >
              <Icon size={18} />
              {tab.label}
            </button>
          );
        })}
      </div>

      {activeTab === "overview" && (
        <section className="admin-stats">
          <div className="admin-stat-card">
            <Users size={25} />
            <h2>{users.length}</h2>
            <p>Registered Users</p>
          </div>

          <div className="admin-stat-card">
            <Search size={25} />
            <h2>{lostItems.length}</h2>
            <p>Lost Reports</p>
          </div>

          <div className="admin-stat-card">
            <Package size={25} />
            <h2>{foundItems.length}</h2>
            <p>Found Reports</p>
          </div>

          <div className="admin-stat-card">
            <ClipboardList size={25} />
            <h2>{claims.length}</h2>
            <p>Total Claims</p>
          </div>
        </section>
      )}

      {activeTab === "users" && (
        <section className="admin-section">
          <h2>Registered Users</h2>

          {users.length === 0 ? (
            <p>No users found.</p>
          ) : (
            <div className="admin-table-wrapper">
              <table className="admin-table">
                <thead>
                  <tr>
                    <th>ID</th>
                    <th>Email</th>
                    <th>Role</th>
                  </tr>
                </thead>

                <tbody>
                  {users.map((user) => (
                    <tr key={user.id}>
                      <td>{user.id}</td>
                      <td>{user.email}</td>
                      <td>
                        <select
                          value={user.role}
                          disabled={
                            user.id === currentUser?.id ||
                            actionLoading === `user-${user.id}`
                          }
                          onChange={(e) =>

function BrowseItems() {
  const [items, setItems] = useState([]);
  const [categories, setCategories] = useState([]);
  const [searchInput, setSearchInput] = useState("");
  const [search, setSearch] = useState("");
  const [selectedCategory, setSelectedCategory] = useState("all");
  const [selectedType, setSelectedType] = useState("found");
  const [page, setPage] = useState(1);
  const [pages, setPages] = useState(1);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    async function loadCategories() {
      try {
        const data = await getCategories();
        setCategories(data);
      } catch (err) {
        console.error("Could not load categories:", err);
      }
    }

    loadCategories();
  }, []);

  useEffect(() => {
    async function loadItems() {
      setLoading(true);
      setError("");

      try {
        const filters = {
          page,
          per_page: 9,
          search,
          category_id: selectedCategory === "all" ? "" : selectedCategory,
        };

        let data;

        if (selectedType === "lost") {
          data = await getLostItems(filters);
        } else if (selectedType === "found") {
          data = await getFoundItems(filters);
        } else {
          const [lostData, foundData] = await Promise.all([
            getLostItems(filters),
            getFoundItems(filters),
          ]);

          const combinedItems = [
            ...(lostData.items || []).map((item) => ({
              ...item,
              report_type: "lost",
            })),
            ...(foundData.items || []).map((item) => ({
              ...item,
              report_type: "found",
            })),
          ];

          data = {
            items: combinedItems,
            pages: Math.max(lostData.pages || 1, foundData.pages || 1),
          };
        }

        setItems(
          (data.items || []).map((item) => ({
            ...item,
            report_type: item.report_type || selectedType,
          }))
        );
        setPages(data.pages || 1);
      } catch (err) {
        setError(err.message || "Could not load items.");
      } finally {
        setLoading(false);
      }
    }

    loadItems();
  }, [page, search, selectedCategory, selectedType]);

  function handleSearch() {
    setPage(1);
    setSearch(searchInput.trim());
  }

  function handleCategoryChange(value) {
    setPage(1);
    setSelectedCategory(value);
  }

  function handleTypeChange(value) {
    setPage(1);
    setSelectedType(value);
  }

  return (
    <div className="browse-page">
      <section className="browse-heading">
        <span>FOUNDIT COLLECTION</span>
        <h1>Browse Items</h1>
        <p>
          Search through lost and found reports to help belongings
          find their way home.
        </p>

        <SearchBar
          value={searchInput}
          onChange={setSearchInput}
          onSearch={handleSearch}
          placeholder="Search for an item..."
        />

        <FilterBar
          categories={categories}
          selectedCategory={selectedCategory}
          onCategoryChange={handleCategoryChange}
          selectedType={selectedType}
          onTypeChange={handleTypeChange}
        />
      </section>

      <section className="browse-results">
        <div className="results-heading">
          <h2>
            {selectedType === "all"
              ? "All Items"
              : selectedType === "lost"
                ? "Lost Items"
                : "Found Items"}
          </h2>
          <span>{items.length} items on this page</span>
        </div>

        {loading && <Loading message="Loading items..." />}

        {!loading && error && (
          <p className="browse-error">{error}</p>
        )}

        {!loading && !error && items.length === 0 && (
          <div className="empty-items">
            <Package size={45} />
            <h3>No items found</h3>
            <p>Try changing your search or filters.</p>
          </div>
        )}

        {!loading && !error && items.length > 0 && (
          <ItemGrid
            items={items}
            type={selectedType === "all" ? "found" : selectedType}
          />
        )}

        {!loading && !error && pages > 1 && (
          <div className="pagination">
            <button
              onClick={() => setPage((current) => current - 1)}
              disabled={page === 1}
            >
              Previous
            </button>

            <span>
              Page {page} of {pages}
            </span>

            <button
              onClick={() => setPage((current) => current + 1)}
              disabled={page === pages}
            >
              Next
            </button>
          </div>
        )}
      </section>
    </div>
  );
}

export default BrowseItems;

import { useEffect, useState } from "react";
import { Bell, Check, CheckCheck } from "lucide-react";
import {
  getNotifications,
  markNotificationRead,
} from "../services/notificationService";

function Notifications() {
  const [notifications, setNotifications] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [updatingId, setUpdatingId] = useState(null);

  useEffect(() => {
    async function loadNotifications() {
      try {
        const data = await getNotifications();
        setNotifications(data);
      } catch (err) {
        setError(err.message || "Could not load notifications.");
      } finally {
        setLoading(false);
      }
    }

    loadNotifications();
  }, []);

  async function handleMarkAsRead(id) {
    setUpdatingId(id);
    setError("");

    try {
      await markNotificationRead(id);

      setNotifications((current) =>
        current.map((notification) =>
          notification.id === id
            ? { ...notification, is_read: true }
            : notification
        )
      );
    } catch (err) {
      setError(err.message || "Could not update notification.");
    } finally {
      setUpdatingId(null);
    }
  }

  const unreadCount = notifications.filter(
    (notification) => !notification.is_read
  ).length;

  if (loading) {
    return <p className="loading-message">Loading notifications...</p>;
  }

  return (
    <main className="page-container notifications-page">
      <div className="notifications-heading">
        <div>
          <h1>Notifications</h1>
          <p>Stay updated on activity related to your reports and claims.</p>
        </div>

        <div className="notification-count">
          <Bell size={20} />
          <span>{unreadCount} unread</span>
        </div>
      </div>

      {error && <p className="form-error">{error}</p>}

      {notifications.length === 0 ? (
        <div className="notifications-empty">
          <Bell size={38} />
          <h2>No notifications yet</h2>
          <p>When there's an update, you'll see it here.</p>
        </div>
      ) : (
        <div className="notifications-list">
          {notifications.map((notification) => (
            <article
              key={notification.id}
              className={`notification-card ${
                notification.is_read ? "read" : "unread"
              }`}
            >
              <div className="notification-icon">
                {notification.is_read ? (
                  <CheckCheck size={21} />
                ) : (
                  <Bell size={21} />
                )}
              </div>

              <div className="notification-content">
                <p>{notification.message}</p>

                <span className="notification-status">
                  {notification.is_read ? "Read" : "Unread"}
                </span>
              </div>

              {!notification.is_read && (
                <button
                  type="button"
                  className="mark-read-button"
                  onClick={() => handleMarkAsRead(notification.id)}
                  disabled={updatingId === notification.id}
                >
                  <Check size={16} />
                  {updatingId === notification.id
                    ? "Updating..."
                    : "Mark as read"}
                </button>
              )}
            </article>
          ))}
        </div>
      )}
    </main>
  );
}

export default Notifications;

import { useState } from "react";
import { Link, useNavigate, useLocation } from "react-router-dom";
import { useAuth } from "../context/AuthContext";

function Signup() {
  const { signupUser } = useAuth();
  const navigate = useNavigate();
  const location = useLocation();

  const redirectTo = location.state?.from || { pathname: "/" };

  const [fullName, setFullName] = useState("");
  const [email, setEmail] = useState("");
  const [institutionalId, setInstitutionalId] = useState("");
  const [universityCode, setUniversityCode] = useState("MOR");
  const [accountType, setAccountType] = useState("student");
  const [password, setPassword] = useState("");
  const [confirmPassword, setConfirmPassword] = useState("");
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);

  async function handleSubmit(event) {
    event.preventDefault();
    setError("");

    if (password.length < 8) {
      setError("Password must be at least 8 characters.");
      return;
    }

    if (new TextEncoder().encode(password).length > 72) {
      setError("Password must not exceed 72 bytes.");
      return;
    }

    if (!/[A-Z]/.test(password)) {
      setError("Password must contain at least one uppercase letter.");
      return;
    }

    if (!/[a-z]/.test(password)) {
      setError("Password must contain at least one lowercase letter.");
      return;
    }

    if (!/[0-9]/.test(password)) {
      setError("Password must contain at least one number.");
      return;
    }

    if (password !== confirmPassword) {
      setError("Passwords do not match.");
      return;
    }

    setLoading(true);

    try {
      await signupUser({
        full_name: fullName,
        email,
        institutional_id: institutionalId,
        university_code: universityCode,
        account_type: accountType,
        password,
      });

      // Go directly to the page the user originally wanted.
      navigate(
        `${redirectTo.pathname || "/"}${redirectTo.search || ""}${redirectTo.hash || ""}`,
        { replace: true }
      );
    } catch (err) {
      setError(err.message || "Could not create your account.");
    } finally {
      setLoading(false);
    }
  }

  return (
    <main className="page-container auth-page">
      <div className="auth-card">
        <h1>Create Account</h1>
        <p>Join FoundIt and help reunite people with their belongings.</p>

        {error && <p className="form-error">{error}</p>}

        <form onSubmit={handleSubmit} className="auth-form">
          <label htmlFor="fullName">Full Name</label>
          <input
            id="fullName"
            type="text"
            value={fullName}
            onChange={(event) => setFullName(event.target.value)}
            placeholder="Enter your full name"
            required
          />

          <label htmlFor="email">Institutional Email Address</label>
          <input
            id="email"
            type="email"
            value={email}
            onChange={(event) => setEmail(event.target.value)}
            placeholder="Enter your university email"
            required
          />

          <label htmlFor="institutionalId">Institutional ID</label>
          <input
            id="institutionalId"
            type="text"
            value={institutionalId}
            onChange={(event) => setInstitutionalId(event.target.value)}
            placeholder="Example: MOR-STU-001"
            required
          />

          <label htmlFor="universityCode">University</label>
          <select
            id="universityCode"
            value={universityCode}
            onChange={(event) => setUniversityCode(event.target.value)}
            required
          >
            <option value="MOR">Moringa School</option>
          </select>

          <label htmlFor="accountType">Account Type</label>
          <select
            id="accountType"
            value={accountType}
            onChange={(event) => setAccountType(event.target.value)}
            required
          >
            <option value="student">Student</option>
            <option value="staff">Staff</option>
          </select>

          <label htmlFor="password">Password</label>

const API_BASE_URL =
  import.meta.env.VITE_API_BASE_URL ||
  "https://foundit-backend-1c64.onrender.com";

export async function apiRequest(endpoint, options = {}) {
  const isFormData = options.body instanceof FormData;

  const response = await fetch(`${API_BASE_URL}${endpoint}`, {
    ...options,
    credentials: "include",
    headers: {
      ...(!isFormData && options.body
        ? { "Content-Type": "application/json" }
        : {}),
      ...options.headers,
    },
  });

  const text = await response.text();

  let data = {};

  try {
    data = text ? JSON.parse(text) : {};
  } catch {
    data = { message: text };
  }

  if (!response.ok) {
    throw new Error(
      data.message || data.error || `Request failed: ${response.status}`
    );
  }

  return data;
}

export default apiRequest;
